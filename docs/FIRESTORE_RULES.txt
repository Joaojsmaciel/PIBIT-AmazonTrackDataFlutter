// Firestore Security Rules para Amazon Track Data
// Regras permissivas para desenvolvimento e testes

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Permitir ler e escrever em todo o banco para usuários autenticados
    match /{document=**} {
      allow read, write: if request.auth != null;
    }
  }
}
